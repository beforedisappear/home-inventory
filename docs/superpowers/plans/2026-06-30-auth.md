# Authorization (email + OTP) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Passwordless email + OTP login with a protected home page, built on the already-shipped HTTP/refresh/token infrastructure.

**Architecture:** A single `services/session` entity owns the whole session lifecycle (`sendCode`, `authenticate`, `me`, `logout`); `features/auth` holds only the two-step `LoginForm` UI, which consumes `sessionQueries`. The `me` query key lives in `kernel`. A pathless guard route protects everything under `/`; `/login` is public. The existing 401 middleware gains a hard redirect to `/login` on refresh failure.

**Tech Stack:** React 19, TanStack Router 1.170 (code-based), TanStack Query 5.101 (`queryOptions`/`mutationOptions`), TanStack Form 1.33, zod 4, HeroUI v3 (react-aria composable), openapi-fetch 0.17, TypeScript ~6.0, Vite 8, bun.

- Import direction is downward only: `pages → features → services → kernel → shared`. Each slice exposes a public API via its `index.ts`.

- Run all `bun`/`tsc`/`vite` commands from `fe/` (`/Users/beforedisappear/dev/home-inventory/fe`). Run all `git` commands from the repo root (`/Users/beforedisappear/dev/home-inventory`).
- `erasableSyntaxOnly: true` is set in both tsconfigs — no runtime `namespace`/`enum`. Query/mutation factories are plain `const` objects (`export const sessionQueries = { me: () => …, … }`), not namespaces.
- openapi-fetch error handling: endpoints whose OpenAPI schema documents no error responses type `error` as `never`, so `if (res.error)` narrows the whole result to `never`. Destructure instead — `const { data, error } = await apiClient…; if (error) throw error; return data!;` — which narrows only the `error` binding. The `data!` assertion is needed because destructuring widens `data` to `T | undefined`.

```ts
import { mutationOptions, queryOptions } from '@tanstack/react-query';

import { sessionKeys } from '@/kernel/session/keys';
import { queryClient } from '@/shared/api/query-client';
import { tokenStorage } from '@/shared/api/token-storage';

import { authenticateRequest } from './authenticate';
import { logoutRequest } from './logout';
import { meRequest } from './me';
import { sendCodeRequest } from './send-code';

// фабрика опций сессии: ключи в kernel, инстансы (apiClient/queryClient) из shared
export const sessionQueries = {
  sendCode: () =>
    mutationOptions({
      mutationFn: sendCodeRequest,
    }),

  authenticate: () =>
    mutationOptions({
      mutationFn: authenticateRequest,
      onSuccess: data => {
        tokenStorage.setTokens(data.accessToken, data.refreshToken);
      },
    }),

  me: () =>
    queryOptions({
      queryKey: sessionKeys.me(),
      queryFn: meRequest,
    }),

  logout: () =>
    mutationOptions({
      mutationKey: ['session', 'logout'],
      mutationFn: logoutRequest,
      onSettled: () => {
        tokenStorage.clear();
        queryClient.clear();
      },
    }),
};
```

```tsx
import { useState } from 'react';

import { useForm } from '@tanstack/react-form';
import { useMutation } from '@tanstack/react-query';
import { useNavigate } from '@tanstack/react-router';
import { z } from 'zod';

import { sessionQueries } from '@/services/session';
import {
  Button,
  ErrorMessage,
  Input,
  Label,
  Spinner,
  TextField,
  toast,
} from '@/shared/ui';

const emailSchema = z.object({
  email: z.string().email('Введите корректный email'),
});

const codeSchema = z.object({
  code: z.string().min(1, 'Введите код'),
});

export function LoginForm() {
  const [step, setStep] = useState<'email' | 'code'>('email');
  const [email, setEmail] = useState('');

  if (step === 'code') {
    return <CodeStep email={email} onBack={() => setStep('email')} />;
  }

  return (
    <EmailStep
      defaultEmail={email}
      onSent={value => {
        setEmail(value);
        setStep('code');
      }}
    />
  );
}

interface EmailStepProps {
  defaultEmail: string;
  onSent: (email: string) => void;
}

function EmailStep(props: EmailStepProps) {
  const { defaultEmail, onSent } = props;

  const sendCode = useMutation(sessionQueries.sendCode());

  const form = useForm({
    defaultValues: { email: defaultEmail },
    validators: { onChange: emailSchema },
    onSubmit: async ({ value }) => {
      try {
        await sendCode.mutateAsync(value);
        toast.success('Код отправлен на почту');
        onSent(value.email);
      } catch {
        toast.danger('Не удалось отправить код');
      }
    },
  });

  return (
    <form
      className='flex flex-col gap-4'
      onSubmit={e => {
        e.preventDefault();
        e.stopPropagation();
        void form.handleSubmit();
      }}
    >
      <form.Field name='email'>
        {field => {
          const showError =
            field.state.meta.isTouched && field.state.meta.errors.length > 0;

          return (
            <TextField
              className='flex flex-col gap-1'
              type='email'
              value={field.state.value}
              onChange={value => field.handleChange(value)}
              onBlur={field.handleBlur}
              isInvalid={showError}
            >
              <Label>Email</Label>
              <Input />
              {showError && (
                <ErrorMessage>
                  {field.state.meta.errors[0]?.message}
                </ErrorMessage>
              )}
            </TextField>
          );
        }}
      </form.Field>

      <form.Subscribe
        selector={state => ({
          canSubmit: state.canSubmit,
          isSubmitting: state.isSubmitting,
        })}
      >
        {({ canSubmit, isSubmitting }) => (
          <Button type='submit' isDisabled={!canSubmit || isSubmitting}>
            {isSubmitting ? <Spinner /> : 'Получить код'}
          </Button>
        )}
      </form.Subscribe>
    </form>
  );
}

interface CodeStepProps {
  email: string;
  onBack: () => void;
}

function CodeStep(props: CodeStepProps) {
  const { email, onBack } = props;

  const navigate = useNavigate();
  const authenticate = useMutation(sessionQueries.authenticate());

  const form = useForm({
    defaultValues: { code: '' },
    validators: { onChange: codeSchema },
    onSubmit: async ({ value }) => {
      try {
        await authenticate.mutateAsync({ email, code: value.code });
        void navigate({ to: '/' });
      } catch {
        toast.danger('Неверный код');
      }
    },
  });

  return (
    <form
      className='flex flex-col gap-4'
      onSubmit={e => {
        e.preventDefault();
        e.stopPropagation();
        void form.handleSubmit();
      }}
    >
      <form.Field name='code'>
        {field => {
          const showError =
            field.state.meta.isTouched && field.state.meta.errors.length > 0;

          return (
            <TextField
              className='flex flex-col gap-1'
              value={field.state.value}
              onChange={value => field.handleChange(value)}
              onBlur={field.handleBlur}
              isInvalid={showError}
            >
              <Label>Код из письма</Label>
              <Input />
              {showError && (
                <ErrorMessage>
                  {field.state.meta.errors[0]?.message}
                </ErrorMessage>
              )}
            </TextField>
          );
        }}
      </form.Field>

      <form.Subscribe
        selector={state => ({
          canSubmit: state.canSubmit,
          isSubmitting: state.isSubmitting,
        })}
      >
        {({ canSubmit, isSubmitting }) => (
          <Button type='submit' isDisabled={!canSubmit || isSubmitting}>
            {isSubmitting ? <Spinner /> : 'Войти'}
          </Button>
        )}
      </form.Subscribe>

      <Button type='button' onPress={onBack}>
        Изменить email
      </Button>
    </form>
  );
}
```

```tsx
import {
  createRootRoute,
  createRoute,
  createRouter,
  redirect,
} from '@tanstack/react-router';

import { HomePage } from '@/pages/home';
import { LoginPage } from '@/pages/login';
import { tokenStorage } from '@/shared/api/token-storage';

import { RootLayout } from '../layouts/root-layout';

export const rootRoute = createRootRoute({
  component: RootLayout,
});

const protectedRoute = createRoute({
  getParentRoute: () => rootRoute,
  id: 'protected',
  beforeLoad: () => {
    if (!tokenStorage.getAccess()) {
      throw redirect({ to: '/login' });
    }
  },
});

const indexRoute = createRoute({
  getParentRoute: () => protectedRoute,
  path: '/',
  component: HomePage,
});

const loginRoute = createRoute({
  getParentRoute: () => rootRoute,
  path: '/login',
  component: LoginPage,
});

const routeTree = rootRoute.addChildren([
  protectedRoute.addChildren([indexRoute]),
  loginRoute,
]);

export const router = createRouter({ routeTree });
```
